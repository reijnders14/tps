/* ============================================================ catalogo.js — Caricamento e rendering del catalogo prodotti ============================================================ */ // Array master dei prodotti caricati dal CSV let prodottiTutti = []; /* ---------- Creazione card prodotto ---------- */ /** * Costruisce il DOM di una card prodotto. * @param {Object} p — prodotto dal CSV * @returns {HTMLElement} */ function creaCard(p) { const card = document.createElement('article'); card.className = 'card-prodotto'; // Wrapper immagine const imgWrap = document.createElement('div'); imgWrap.className = 'card-img-wrap'; const img = document.createElement('img'); img.src = `assets/${p.immagine}`; img.alt = p.nome; img.loading = 'lazy'; img.onerror = () => { // Placeholder visivo se l'immagine non esiste imgWrap.style.background = 'linear-gradient(135deg, #ede5d4, #c9b99a)'; img.style.opacity = '0'; }; // Badge categoria const badge = document.createElement('span'); badge.className = 'card-badge'; badge.textContent = p.categoria; imgWrap.appendChild(img); imgWrap.appendChild(badge); // Corpo card const body = document.createElement('div'); body.className = 'card-body'; const marca = document.createElement('p'); marca.className = 'card-marca'; marca.textContent = p.marca; const nome = document.createElement('h2'); nome.className = 'card-nome'; nome.textContent = p.nome; const desc = document.createElement('p'); desc.className = 'card-desc'; desc.textContent = p.descrizione_breve; body.appendChild(marca); body.appendChild(nome); body.appendChild(desc); // Footer card con prezzo e pulsante const footer = document.createElement('div'); footer.className = 'card-footer'; const prezzo = document.createElement('div'); prezzo.className = 'card-prezzo'; const sup = document.createElement('sup'); sup.textContent = '€'; prezzo.appendChild(sup); prezzo.appendChild(document.createTextNode( parseFloat(p.prezzo).toFixed(2).replace('.', ',') )); const btnDettagli = document.createElement('a'); btnDettagli.href = `prodotto.html?id=${p.id}`; btnDettagli.className = 'btn btn-primary'; btnDettagli.textContent = 'Dettagli'; btnDettagli.setAttribute('aria-label', `Dettagli di ${p.nome}`); footer.appendChild(prezzo); footer.appendChild(btnDettagli); card.appendChild(imgWrap); card.appendChild(body); card.appendChild(footer); return card; } /* ---------- Rendering griglia ---------- */ /** * Svuota e ri-popola la griglia con i prodotti forniti. * @param {Array
